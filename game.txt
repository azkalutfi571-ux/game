#include <ncurses.h>
#include <cstdlib>
#include <ctime>
#include <deque>

using namespace std;

const int HEIGHT = 20;
const int WIDTH = 50;

struct Point {
    int y;
    int x;
};

deque<Point> snake;
Point food;

int score = 0;
int direction = KEY_RIGHT;
bool gameOver = false;
bool paused = false;

bool samePoint(Point a, Point b) {
    return a.y == b.y && a.x == b.x;
}

void placeFood() {
    bool valid = false;

    while (!valid) {
        food.y = rand() % HEIGHT;
        food.x = rand() % WIDTH;
        valid = true;

        for (int i = 0; i < (int)snake.size(); i++) {
            if (samePoint(food, snake[i])) {
                valid = false;
                break;
            }
        }
    }
}
